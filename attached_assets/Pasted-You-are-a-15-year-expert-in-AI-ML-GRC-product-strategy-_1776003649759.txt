  CREATE TABLE evidence_chain (
    id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id          uuid REFERENCES organizations(id),
    sequence_num    bigint NOT NULL, -- global sequence per org
    evidence_id     uuid REFERENCES evidence(id),
    entity_type     text NOT NULL, -- control|policy|risk|audit|incident|hitl
    entity_id       uuid NOT NULL,
    action          text NOT NULL, -- create|update|approve|reject|sign|verify
    actor_id        uuid REFERENCES user_profiles(id),
    actor_name      text,
    payload_hash    text NOT NULL, -- SHA-256 of canonical JSON payload
    prev_hash       text,          -- hash of previous chain entry
    chain_hash      text NOT NULL, -- SHA-256(payload_hash + prev_hash)
    payload         jsonb NOT NULL, -- full snapshot at time of action
    signed_at       timestamptz DEFAULT now(),
    -- Immutable: no UPDATE/DELETE RLS policy
    UNIQUE (org_id, sequence_num)
  );
  -- RLS: SELECT only (append-only, no UPDATE/DELETE)

Hash calculation (client-side utility): src/lib/evidenceChain.ts
  import { sha256 } from "@noble/hashes/sha256"
  import { bytesToHex } from "@noble/hashes/utils"

  export async function appendToChain(
    orgId: string,
    entityType: string,
    entityId: string,
    action: string,
    payload: object,
    actorId: string,
    actorName: string
  ) {
    // 1. Get last chain entry for this org
    const { data: last } = await supabase
      .from('evidence_chain')
      .select('chain_hash, sequence_num')
      .eq('org_id', orgId)
      .order('sequence_num', { ascending: false })
      .limit(1).single()

    // 2. Calculate hashes
    const canonical = JSON.stringify(payload, Object.keys(payload).sort())
    const payloadHash = bytesToHex(sha256(new TextEncoder().encode(canonical)))
    const prevHash = last?.chain_hash ?? '0'.repeat(64)
    const chainHash = bytesToHex(
      sha256(new TextEncoder().encode(payloadHash + prevHash))
    )

    // 3. Insert to chain
    await supabase.from('evidence_chain').insert({
      org_id: orgId,
      sequence_num: (last?.sequence_num ?? 0) + 1,
      entity_type: entityType,
      entity_id: entityId,
      action,
      actor_id: actorId,
      actor_name: actorName,
      payload_hash: payloadHash,
      prev_hash: prevHash,
      chain_hash: chainHash,
      payload,
    })
  }

  // Verify chain integrity
  export async function verifyChain(orgId: string): Promise<{
    valid: boolean; entries: number; brokenAt?: number
  }> {
    const { data: entries } = await supabase
      .from('evidence_chain')
      .select('*')
      .eq('org_id', orgId)
      .order('sequence_num')

    let prevHash = '0'.repeat(64)
    for (const entry of entries ?? []) {
      const canonical = JSON.stringify(entry.payload, Object.keys(entry.payload).sort())
      const payloadHash = bytesToHex(sha256(new TextEncoder().encode(canonical)))
      const expectedChain = bytesToHex(
        sha256(new TextEncoder().encode(payloadHash + prevHash))
      )
      if (expectedChain !== entry.chain_hash) {
        return { valid: false, entries: entries!.length, brokenAt: entry.sequence_num }
      }
      prevHash = entry.chain_hash
    }
    return { valid: true, entries: entries?.length ?? 0 }
  }

Feature gate hook: src/hooks/useFeatureGate.ts
  export function useFeature(key: string): {
    enabled: boolean; tier: 'open_core'|'enterprise'|'add_on'; gateable: boolean
  } {
    const { license } = useLicenseStore()
    const feature = license.features[key]
    if (!feature) return { enabled: false, tier: 'enterprise', gateable: true }
    if (feature.tier === 'open_core') return { enabled: true, tier: 'open_core', gateable: false }
    return {
      enabled: license.plan === 'enterprise' || license.plan === 'trial',
      tier: feature.tier,
      gateable: true
    }
  }
